use std::collections::{HashSet, VecDeque};

use itertools::Itertools;

use crate::{dot, read, DIRS_EXTRA};

/// (y, x, prev)
struct Point(i32, i32, HashSet<(i32, i32)>);

pub fn run() {
    let file = read!()
        .map(|l| {
            l.chars()
                .map(|c| match c {
                    'X' => 0,
                    'M' => 1,
                    'A' => 2,
                    'S' => 3,
                    _ => panic!(),
                })
                .collect_vec()
        })
        .collect_vec();

    let mut interests = vec![];

    for (y, r) in file.iter().enumerate() {
        for (x, &c) in r.iter().enumerate() {
            if c == 0 {
                // X
                interests.push((y as i32, x as i32));
            }
        }
    }

    let mut sum = 0;
    let mut xset = HashSet::new();
    for (y, x) in interests {
        sum += solve(&file, (y, x), &mut xset);
    }

    let arr = ['X', 'M', 'A', 'S'];

    dot!(file, |y, x, c| if xset.contains(&(y, x)) {
        arr[c as usize]
    } else {
        '.'
    });

    println!("Part1: {sum}");
}

fn solve(grid: &Vec<Vec<i32>>, (sy, sx): (i32, i32), points: &mut HashSet<(i32, i32)>) -> i32 {
    // let mut visited: HashMap<Point, Option<Point>> = HashMap::new();

    let mut q = VecDeque::new();

    q.push_front(Point(sy, sx, HashSet::from_iter([(sy, sx)])));

    let mut count = 0;

    loop {
        let Some(Point(y, x, prev)) = q.pop_back() else {
            break;
        };

        let curr = grid[y as usize][x as usize];

        if curr == 3 {
            count += 1;
            points.extend(&prev);
        } else if prev.len() < 4 {
            for (dy, dx) in DIRS_EXTRA {
                let (ny, nx) = (y + dy, x + dx);
                if prev.contains(&(ny, nx))
                    || ny < 0
                    || nx < 0
                    || ny >= grid.len() as i32
                    || nx >= grid[0].len() as i32
                {
                    continue;
                }

                if grid[ny as usize][nx as usize] - curr != 1 {
                    continue;
                }

                let mut next_prev = prev.clone();
                next_prev.insert((ny, nx));

                q.push_front(Point(ny, nx, next_prev));
            }
        }
    }

    count
}
